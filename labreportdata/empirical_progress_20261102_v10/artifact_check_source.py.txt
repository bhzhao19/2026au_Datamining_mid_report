"""Verify this measured report's source bindings and displayed condition table."""
from pathlib import Path
from html.parser import HTMLParser
from collections import Counter
from datetime import datetime, timezone
import hashlib
import json
import subprocess
import xml.etree.ElementTree as ET
from PIL import Image

OUT = Path(__file__).resolve().parent
ROOT = OUT.parents[1]

def digest(path):
    return hashlib.sha256(Path(path).read_bytes()).hexdigest()

def read(path):
    return json.loads(Path(path).read_text())

class Tables(HTMLParser):
    def __init__(self):
        super().__init__()
        self.tables, self.table, self.row, self.cell = [], None, None, None
    def handle_starttag(self, tag, attrs):
        if tag == 'table': self.table = []
        elif tag == 'tr' and self.table is not None: self.row = []
        elif tag in ('td', 'th') and self.row is not None: self.cell = ''
    def handle_data(self, data):
        if self.cell is not None: self.cell += data
    def handle_endtag(self, tag):
        if tag in ('td', 'th') and self.cell is not None:
            self.row.append(self.cell); self.cell = None
        elif tag == 'tr' and self.row is not None:
            self.table.append(self.row); self.row = None
        elif tag == 'table' and self.table is not None:
            self.tables.append(self.table); self.table = None

manifest = read(OUT/'manifest.json')
table = read(OUT/'cpu_results_table.json')
assert manifest['final_project_complete'] is False
assert table['full_project_complete'] is False
assert len(manifest['figures']) == 26
assert len(manifest['cpu_conditions']) == len(table['rows']) == 8
assert digest(OUT/'report.pdf') == manifest['report_pdf_sha256']
assert digest(OUT/'report.html') == manifest['report_html_sha256']
assert digest(OUT/'cpu_results_table.json') == manifest['cpu_results_table_sha256']
assert table['source_conditions'] == manifest['cpu_conditions']
exports = 0
for stem, info in manifest['figures'].items():
    for suffix in ('png','pdf','svg'):
        path = OUT/'figures'/f'{stem}.{suffix}'
        assert digest(path) == info[f'{suffix}_sha256']
        if suffix == 'png':
            with Image.open(path) as im: im.verify()
        elif suffix == 'svg':
            assert ET.parse(path).getroot().tag.endswith('svg')
        else:
            assert path.read_bytes().startswith(b'%PDF-')
        exports += 1
labels = {'test_iid':'IID', 'test_ood_scale':'规模 OOD',
          'test_ood_representation':'表示 OOD', 'test_ood':'组合 OOD'}
calibrations = {}
dataset_counts = Counter()
for item, cells in zip(manifest['cpu_conditions'],table['rows']):
    rp, ap = ROOT/item['report'], ROOT/item['audit']
    assert digest(rp) == item['report_sha256']
    assert digest(ap) == item['audit_sha256']
    result, audit = read(rp), read(ap)
    assert audit['passed'] is True and audit['report_sha256'] == digest(rp)
    assert result['condition_complete'] is True and result['full_project_complete'] is False
    assert result['rows'] == audit['rows_checked'] == 400 and audit['groups'] == 200
    assert item['dataset'] == result['dataset'] and item['condition'] == result['condition']
    seed = int(result['dataset'].rsplit('seed',1)[1])
    tp = ROOT/f'results/cpu_secondary_20261102/early_calibration/seed{seed}.temperature.json'
    temp = read(tp)
    assert digest(tp) == result['calibration_sha256']
    assert temp['temperature'] == result['temperature']
    frozen = {'temperature':temp['temperature'],'artifact_sha256':digest(tp)}
    assert str(seed) not in calibrations or calibrations[str(seed)] == frozen
    calibrations[str(seed)] = frozen
    before, after = audit['metrics']['before'], audit['metrics']['after']
    lo, hi = result['accuracy_group_bootstrap']['ci95']
    expected = [seed, labels[item['condition']], f"{before['accuracy']*100:.2f}",
                f'[{lo*100:.2f}, {hi*100:.2f}]',
                *[f'{before[m]:.4f} → {after[m]:.4f}' for m in ('ece','nll','brier')],
                f"{audit['error_auroc_raw']:.3f} → {audit['error_auroc_calibrated_canonical']:.3f}"]
    assert cells == expected, (cells,expected)
    assert before['accuracy'] == after['accuracy']
    dataset_counts[item['dataset']] += 1
assert calibrations == table['frozen_calibrations'] == manifest['cpu_frozen_calibrations']
html = (OUT/'report.html').read_text()
parser = Tables(); parser.feed(html)
matching = [t for t in parser.tables if t and t[0] == table['headers']]
assert len(matching) == 1
assert matching[0][1:] == [[str(c) for c in row] for row in table['rows']]
assert '8 / 12' in html and 'GPU 主实验 E1–E4' in html
assert '真实 GPU/API 交接与最终报告仍待完成' in html
error_table = read(OUT/'cpu_error_uncertainty_table.json')
uncertainty_dir = ROOT/'results/cpu_error_uncertainty_20261102_v3'
uncertainty = read(uncertainty_dir/'summary.json')
uncertainty_proof = read(uncertainty_dir/'verification.json')
assert uncertainty_proof['passed'] and uncertainty_proof['conditions_checked']==8
assert digest(uncertainty_dir/'summary.json') == manifest['cpu_error_uncertainty_sha256'] == error_table['summary_sha256']
assert digest(uncertainty_dir/'verification.json') == manifest['cpu_error_verification_sha256'] == error_table['verification_sha256']
assert digest(OUT/'cpu_error_uncertainty_table.json') == manifest['cpu_error_uncertainty_table_sha256']
assert error_table['posthoc'] and error_table['additional_independent_samples']==0
by_condition = {(r['dataset'],r['condition']):r for r in uncertainty['conditions']}
expected_error_rows=[]
for item in manifest['cpu_conditions']:
    row=by_condition[(item['dataset'],item['condition'])]
    assert row['report_sha256']==item['report_sha256'] and row['audit_sha256']==item['audit_sha256']
    cells=[int(item['dataset'].rsplit('seed',1)[1]),labels[item['condition']]]
    stats=row['statistics']
    for key in ('raw','calibrated','difference'):
        lo,hi=stats['ci95'][key]
        cells.append(f"{stats['point'][key]:.4f} [{lo:.4f}, {hi:.4f}]")
    expected_error_rows.append(cells)
assert error_table['rows']==expected_error_rows
error_html=[t for t in parser.tables if t and t[0]==error_table['headers']]
assert len(error_html)==1 and error_html[0][1:]==[[str(c) for c in row] for row in expected_error_rows]
assert '权重和温度工件保持固定' in html

    'kind':'measured-progress-report-artifact-verification', 'passed':True,

    'required_data_seeds':3,'pdf_summary_table_rows_independently_recomputed':8,
    'table_check_method':'All JSON/HTML table cells independently compared to audit metrics, original group CIs and exact per-seed calibration artifacts; PDF table visually checked',
    'table_sha256':digest(OUT/'cpu_results_table.json'),
    'error_uncertainty_conditions_checked':8, 'error_uncertainty_intervals_checked':24,
    'error_uncertainty_table_sha256':digest(OUT/'cpu_error_uncertainty_table.json'),
    'posthoc_order_diagnostic_included':True,
    'baseline_audit_exact_bindings_verified':True,
    'baseline_artifact_hashes_checked':len(audit['artifact_sha256']),

    'manifest_sha256':digest(OUT/'manifest.json'),
    'generator_source_archive':str(source.relative_to(ROOT)),
    'generator_source_sha256':digest(source),
    'artifact_checker_source_sha256':digest(__file__),
    'visually_checked':['PDF page2 LabQueue policy priority and fitting roles',

    'scope':'Eight of twelve supplementary CPU conditions; original GPU E1-E4, public LLM, actual GPU/API and final-report requirements remain pending'
}
output = OUT/'artifact_verification.json'
assert not output.exists()

print(json.dumps({k:proof[k] for k in ('passed','pages','figures','figure_exports_checked','cpu_conditions_checked')}))
