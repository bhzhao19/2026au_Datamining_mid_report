def load_uniform_reference(path, conditions):
    if not path:return None
    reference=loads(Path(path).read_text())
    if (reference.get("kind")!="analytical-fixed-uniform-labqueue-reference" or
            reference.get("passed") is not True or reference.get("additional_model_calls")!=0 or
            reference.get("additional_independent_samples")!=0 or reference.get("fitted_parameters")!=0 or
            reference["analysis_source_sha256"]!=sha256(Path(path).parent/"analysis_source.py.txt")):
        raise ValueError("Bound analytical uniform reference required")
    by_key={(r["dataset"],r["condition"]):r for r in reference["conditions"]}
    expected={(c["dataset"],c["condition"]):c for _,c,_,_,_ in conditions}
    if len(by_key)!=12 or set(by_key)!=set(expected):
        raise ValueError("Uniform reference must cover exactly all12 displayed conditions")
    for key,item in by_key.items():
        path=ROOT/item["data"];rows=read_jsonl(path)
        labels=sorted(rows[0]["questions"]["action"]["criteria"])
        counts=Counter(r["targets"]["action"]["label"] for r in rows)
        if (sha256(path)!=item["data_sha256"] or item["data_sha256"]!=expected[key]["data_sha256"] or
                len(rows)!=400 or len({r["group"] for r in rows})!=200 or len(labels)!=4 or
                counts!=item["label_counts"] or set(counts)!=set(labels) or set(counts.values())!={100} or
                item["classes"]!=labels or item["fixed_tie_label"]!=labels[0] or
                any(sorted(r["questions"]["action"]["criteria"])!=labels for r in rows)):
            raise ValueError("Uniform reference is not the declared balanced four-class cohort")
        values={"accuracy":.25,"ece":0.,"brier":.75,"nll":math.log(4),"error_detection_auroc":.5,"confidence":.25}
        if any(not math.isclose(item[k],v,rel_tol=0,abs_tol=1e-12) for k,v in values.items()):
            raise ValueError("Uniform analytical values changed")
    return reference


def load_error_uncertainty(summary_path, proof_path, conditions):
    """Require an audited interval bundle for exactly the displayed conditions."""
    if bool(summary_path) != bool(proof_path):
        raise ValueError("Error uncertainty and independent verification must be supplied together")
    if not summary_path:return None
    summary, proof = loads(Path(summary_path).read_text()), loads(Path(proof_path).read_text())
    if (summary.get("kind") != "posthoc-paired-cpu-error-auroc-uncertainty" or not summary.get("posthoc") or
            summary.get("additional_independent_samples") != 0 or summary.get("full_project_complete") is not False or
            proof.get("kind") != "independent-cpu-error-uncertainty-audit" or proof.get("passed") is not True or
            proof["summary_sha256"] != sha256(summary_path)):
        raise ValueError("Verified post-hoc CPU interval evidence required")
    base = Path(summary_path).parent
    if (sha256(base/"bootstrap_replicates.jsonl") != summary["bootstrap_replicates_sha256"] or
            proof["trace_sha256"] != summary["bootstrap_replicates_sha256"] or
            sha256(base/"render_manifest.json") != proof["render_manifest_sha256"]):
        raise ValueError("Uncertainty trace/render bindings changed")
    for name,digest in proof["exports_sha256"].items():
        if Path(name).name != name or sha256(base/name) != digest:
            raise ValueError("Uncertainty export changed")
    expected = {(c["dataset"],c["condition"]):(c,a,p,ap) for _,c,a,p,ap in conditions}
    by_key = {(c["dataset"],c["condition"]):c for c in summary["conditions"]}
    if (len(by_key) != len(summary["conditions"]) or set(by_key) != set(expected) or
            proof["conditions_checked"] != len(expected)):
        raise ValueError("Uncertainty must cover exactly the displayed complete CPU conditions")
    if (summary["bootstrap"]["unit"] != "scenario group" or summary["bootstrap"]["replicates"] != 1000 or
            summary["bootstrap"]["seed"] != 17):
        raise ValueError("Wrong paired uncertainty protocol")
    for key,(c,a,p,ap) in expected.items():
        item=by_key[key]
        if (item["report_sha256"] != sha256(p) or item["audit_sha256"] != sha256(ap) or
                any(item[k] != c[k] for k in ("data_sha256","predictions_sha256","calibration_sha256","temperature")) or
                summary["source_fingerprint"] != fingerprint(c["model"])):
            raise ValueError("Uncertainty uses different predictions/calibration/model")
        stats=item["statistics"]
        if stats["defined"]:
            if (not math.isclose(stats["point"]["raw"],a["error_auroc_raw"],abs_tol=1e-12) or
                    not math.isclose(stats["point"]["calibrated"],a["error_auroc_calibrated_canonical"],abs_tol=1e-12) or
                    not math.isclose(stats["point"]["difference"],stats["point"]["calibrated"]-stats["point"]["raw"],abs_tol=1e-12)):
                raise ValueError("Uncertainty point differs from condition audit")
            for metric in ("raw","calibrated","difference"):
                lo,hi=stats["ci95"][metric]
                if not all(math.isfinite(v) for v in (lo,hi)) or lo>hi:
                    raise ValueError("Invalid uncertainty interval")
        elif a["error_auroc_raw"] is not None or a["error_auroc_calibrated_canonical"] is not None:
            raise ValueError("Undefined interval disagrees with condition audit")
    return [by_key[(c["dataset"],c["condition"])] for _,c,_,_,_ in conditions]
